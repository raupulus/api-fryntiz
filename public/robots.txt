# Lo que no debe salir en un buscador.
#
# Antes este fichero era un `Disallow:` vacío, o sea «indexa todo»: el panel de
# administración, el de usuario, los formularios de sesión y las páginas de
# gestión de la newsletter —que llevan un token de un solo uso en la URL—
# estaban abiertos a cualquier rastreador.

User-agent: *

# Paneles de administración y de usuario.
#
# Ojo: esto pide que NO SE RASTREEN, que no es lo mismo que no indexar. Un
# buscador que llegue por un enlace externo puede listar la URL sin abrir la
# página, y este mismo Disallow le impide leer la directiva de dentro. Quien
# impide la indexación de verdad es la cabecera `X-Robots-Tag: noindex` que
# pone `App\Http\Middleware\NoIndex` en los dos paneles.
Disallow: /admin
Disallow: /panel
Disallow: /livewire

# Sesión y cuenta.
Disallow: /login
Disallow: /register
Disallow: /user/
Disallow: /two-factor-challenge

# URLs con token: no son páginas, son enlaces personales de un solo uso.
Disallow: /newsletter/
Disallow: /cv/s/

# La API se consume con un cliente, no se lee en un buscador.
#
# /docs no se lista aquí a propósito, pero no hace falta: exige sesión
# iniciada (`routes/web.php`), así que un rastreador anónimo sólo ve la
# redirección al login y nunca llega al contenido. No indexarlo es
# consecuencia de la autenticación, no de este fichero.
Disallow: /api/

# El sitemap vive en ESTE mismo host (los vhosts de `docs/deploys/vhosts/`
# sirven la aplicación en `api.raupulus.dev`, no en `raupulus.dev`): apuntar a
# otro dominio deja el sitemap sin descubrir para cualquier rastreador que siga
# esta directiva.
Sitemap: https://api.raupulus.dev/sitemap.xml
