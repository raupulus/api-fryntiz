Python

Aemet OpenData
Ejemplo en Python http.clien 

import http.client
import urllib.parse

API_KEY = "TU_API_KEY"

params = urllib.parse.urlencode({
    "api_key": API_KEY
})

conn = http.client.HTTPSConnection(
    "opendata.aemet.es",
    timeout=30
)

conn.request(
    "GET",
    f"/opendata/api/valores/climatologicos/inventarioestaciones/todasestaciones/?{params}"
)

response = conn.getresponse()

if response.status == 200:
    print(response.read().decode("utf-8"))
else:
    print(f"Error HTTP {response.status}: {response.reason}")

conn.close()
